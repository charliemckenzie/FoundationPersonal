# Component Status

Full inventory of every component in Foundation. The data lives in `component-status.ts` (a single source of truth shared with the codemap generator); counts are computed from it so they stay accurate as the library grows. Sort by name or status.

export const ComponentTable = () => {
  const [sortCol, setSortCol] = useState('name');
  const [sortDir, setSortDir] = useState('asc');

  const toggle = (col) => {
    if (sortCol === col) setSortDir(d => d === 'asc' ? 'desc' : 'asc');
    else { setSortCol(col); setSortDir('asc'); }
  };

  const sorted = [...COMPONENTS].sort((a, b) => {
    const cmp = sortCol === 'name'
      ? a.name.localeCompare(b.name)
      : STATUS_ORDER[a.status] - STATUS_ORDER[b.status];
    return sortDir === 'asc' ? cmp : -cmp;
  });

  const indicator = (col) => sortCol === col ? (sortDir === 'asc' ? ' ▲' : ' ▼') : ' ⇅';
  const thStyle = { cursor: 'pointer', userSelect: 'none', whiteSpace: 'nowrap' };

  return (
    <table>
      <thead>
        <tr>
          <th style={thStyle} onClick={() => toggle('name')}>Component{indicator('name')}</th>
          <th style={thStyle} onClick={() => toggle('status')}>Status{indicator('status')}</th>
          <th>Story</th>
        </tr>
      </thead>
      <tbody>
        {sorted.map(({ name, status, story }) => (
          <tr key={name}>
            <td>{name}</td>
            <td>
              <span style={{display:'inline-block',width:'10px',height:'10px',borderRadius:'50%',backgroundColor:STATUS_COLOR[status],marginRight:'6px',verticalAlign:'middle'}} aria-hidden="true" />
              <code>{status}</code>
            </td>
            <td>{story ? <a href={story}>{name}</a> : <span style={{color:'#999',fontStyle:'italic'}}>No story yet</span>}</td>
          </tr>
        ))}
      </tbody>
    </table>
  );
};
